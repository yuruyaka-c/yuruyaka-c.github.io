## 15.1

### 15.1.1

```c title="番兵法の例"
#include <stdio.h>

void PrintIntArraySentinel(const int arr[])
{
	putchar('[');

	for (int i = 0; ; ++i)
	{
		if (arr[i] == -1) // 番兵に到達したら終了
		{
			break;
		}

		printf(((i == 0) ? "%d" : ", %d"), arr[i]);
	}

	puts("]");
}

int main()
{
	int scores[] = { 80, 100, 65, 90, 75, -1 }; // -1 を番兵として利用
	PrintIntArraySentinel(scores);
}
```

### 15.1.5

```c title="文字配列での番兵法の例"
#include <stdio.h>

void PrintCharArraySentinel(const char s[])
{
	for (int i = 0; ; ++i)
	{
		if (s[i] == '\0') // ヌル文字に到達したら終了
		{
			break;
		}

		putchar(s[i]);
	}

	putchar('\n');
}

int main()
{
	// すべての要素を '\0'（0）で初期化
	char s[11] = { };
	s[0] = 'A';
	s[1] = 'p';
	s[2] = 'p';
	s[3] = 'l';
	s[4] = 'e';
	s[5] = '\0';
	PrintCharArraySentinel(s);

	s[3] = '\0';
	PrintCharArraySentinel(s);

	s[1] = '\0';
	PrintCharArraySentinel(s);

	s[1] = 'm';
	s[2] = 'e';
	s[3] = 'r';
	s[4] = 'i';
	s[5] = 'c';
	s[6] = 'a';
	s[7] = '\0';
	PrintCharArraySentinel(s);
}
```

```c title="ヌル終端文字列の要素を確認する例"
#include <stdio.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

void PrintCharArray(const char s[], int count)
{
	for (int i = 0; i < count; ++i)
	{
		printf("s[%d] = '%c' (ASCII: %d)\n", i, s[i], s[i]);
	}

	putchar('\n');
}

int main()
{
	const char s1[] = "Apple";
	puts("s1:");
	PrintCharArray(s1, COUNTOF(s1));

	const char s2[] = "Banana\n";
	puts("s2:");
	PrintCharArray(s2, COUNTOF(s2));
}
```

### 15.2.5

```c title="ヌル終端と標準ライブラリ関数"
#include <stdio.h>

int main()
{
	// ヌル終端されていない
	// s1 = { ?, ?, ?, ... }
	char s1[101];

	// ヌル終端されている
	// s2 = { 'c', 'a', 't', '\0', '\0', '\0', ... }
	char s2[101] = { 'c', 'a', 't' };
	puts(s2); // 問題なく動作

	// この時点ではヌル終端されていない
	char s3[101];

	s3[0] = 'd';
	s3[1] = 'o';
	s3[2] = 'g';
	s3[3] = '\0'; // ヌル終端を明示的に設定し、ヌル終端文字列にする

	puts(s3); // 問題なく動作
}
```

### 15.2.6

```c title="文字列を標準入力から読み込む関数の例"
#include <stdio.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

void ScanString(char s[], int maxCount)
{
	int i = 0;

	for (; i < (maxCount - 1); ++i)
	{
		const int ch = getchar();

		if ((ch == '\n') || (ch == EOF)) // 改行または EOF で終了
		{
			break;
		}
		else
		{
			s[i] = (char)ch;
		}
	}

	s[i] = '\0'; // ヌル終端を設定
}

int main()
{
	printf("文字列を入力してください > ");
	char s[101];
	ScanString(s, COUNTOF(s));

	printf("入力された文字列: %s\n", s);
}
```

```c title="ヌル終端文字列の内容を比較する例"
#include <stdio.h>

bool StrEquals(const char a[], const char b[])
{
	for (int i = 0; ; ++i)
	{
		if (a[i] != b[i]) // どこか 1 文字でも違えば不一致
		{
			return false;
		}

		if (a[i] == '\0') // ここまですべて一致していて '\0' なら同じ文字列
		{
			return true;
		}
	}
}

int main()
{
	const char s1[] = "Apple";
	const char s2[] = "Apple";
	const char s3[] = "App";

	if (StrEquals(s1, s2))
	{
		puts("s1 と s2 は等しい");
	}
	else
	{
		puts("s1 と s2 は異なる");
	}

	if (StrEquals(s1, s3))
	{
		puts("s1 と s3 は等しい");
	}
	else
	{
		puts("s1 と s3 は異なる");
	}
}
```

```c title="自作の strlen 関数の例"
#include <stdio.h>

size_t Strlen(const char s[])
{
	size_t length = 0;

	for (int i = 0; ; ++i)
	{
		if (s[i] == '\0') // ヌル文字に到達したら終了
		{
			break;
		}

		++length;
	}

	return length;
}

int main()
{
	const char s1[] = "Apple";
	const char s2[] = "Banana\n";
	printf("s1 の長さ: %zu\n", Strlen(s1));
	printf("s2 の長さ: %zu\n", Strlen(s2));
}
```

```c title="自作の strcmp 関数の例"
#include <stdio.h>

int Strcmp(const char s1[], const char s2[])
{
	for (int i = 0; ; ++i)
	{
		if (s1[i] != s2[i])
		{
			if (s1[i] < s2[i])
			{
				return -1; // s1 が小さい
			}
			else
			{
				return 1; // s1 が大きい
			}
		}

		if (s1[i] == '\0') // 両方ともヌル文字に到達したら終了
		{
			break;
		}
	}

	return 0; // 同じ内容
}

int main()
{
	const char s1[] = "Apple";
	const char s2[] = "Banana";
	const char s3[] = "Apple";
	printf("s1 と s2 の比較結果: %d\n", Strcmp(s1, s2));
	printf("s1 と s3 の比較結果: %d\n", Strcmp(s1, s3));
	printf("s2 と s3 の比較結果: %d\n", Strcmp(s2, s3));
}
```
