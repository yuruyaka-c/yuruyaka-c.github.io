    配列の要素へのアクセス例

int main()
{
	int scores[3]; // 要素数 3 の int 型配列を定義

	// 各要素に値を代入
	scores[0] = 80;
	scores[1] = 100;
	scores[2] = 65;

	printf("2 人目: %d\n", scores[1]);
	printf("3 人目: %d\n", scores[2]);
}

    ⚠️ 配列の要素数に定数式以外を使用している
    #include <stdio.h>

int GetSize()
{
	return 10;

{

    （参考）⭕️ 動的なメモリ確保の例（入力した要素数で配列を確保）
    #include <stdio.h>
#include <stdlib.h>

int main()
{
	int n;
	printf("配列の要素数を入力してください > ");
	scanf("%d", &n);

	int* a = (int*)malloc(n * sizeof(int)); // 動的なメモリ確保

	if (a == NULL)
	{
		return 0;
	}

	for (int i = 0; i < n; ++i)
	{
		a[i] = i * 10;
		printf("a[%d] = %d\n", i, a[i]);
	}

	free(a); // 確保したメモリの解放
}

    14.1.9¶

    ⚠️ 「==」で比較しても「内容比較」にはならない
    #include <stdio.h>

int main()
{
	int a[3] = { 10, 20, 30 };
	int b[3] = { 10, 20, 30 };

	if (a == b) // 先頭要素のアドレスどうしを比較している
	{
		puts("a と b は同じ領域");
	}
	else
	{
		puts("a と b は別の領域");
	}
}
    配列の内容を比較する例
    #include <stdio.h>

int main()
{
	int a[3] = { 10, 20, 30 };
	int b[3] = { 10, 20, 30 };

	bool equal = true;

	for (int i = 0; i < 3; ++i)
	{
		if (a[i] != b[i])
		{
			equal = false;
			break;
		}
	}

	if (equal)
	{
		puts("配列 a と b は等しい");
	}
	else
	{
		puts("配列 a と b は等しくない");
	}
}

    配列の要素数を使ったループ処理例（配列の全要素の合計と平均の出力）
    #include <stdio.h>

int main()

    14.5.3¶

    配列の要素の最大値を求める関数の例
    #include <stdio.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

int Max(const int arr[], int count)
{
	int max = arr[0]; // 先頭の要素を仮の最大値とする

	for (int i = 1; i < count; ++i)
	{
		if (max < arr[i])
		{
			max = arr[i];
		}
	}

	return max;
}

int main()
{
	const int scores[] = { 80, 100, 65, 90, 75 };
	printf("最大値: %d\n", Max(scores, COUNTOF(scores)));
}

    14.5.7¶

    配列の要素が昇順に並んでいるかを判定する関数の例
    #include <stdio.h>

bool IsSorted(const int arr[], int count)
{
	for (int i = 1; i < count; ++i)

		}

}

int main()

	int scores[5] = { 80, 100, 65, 90, 75 };
	Fill((scores + 2), (COUNTOF(scores) - 2), 0); // ポインタ演算を使用
	PrintIntArray(scores, COUNTOF(scores));

    アルファベットの出現回数を数えるプログラム
    #include <stdio.h>
#include <ctype.h>

int main()
{
	// 各アルファベットの出現回数を格納する配列
	// counts[0] が 'A'/'a' の出現回数
	// counts[1] が 'B'/'b' の出現回数
	// ...
	// counts[25] が 'Z'/'z' の出現回数
	int counts[26] = { };

	for (;;)
	{
		const int c = getchar();

		// 改行が来たら終了する
		if (c == '\n')
		{
			break;
		}

		if (isalpha(c))
		{
			// 小文字にそろえる
			const int lower = tolower(c);

			// 'a'～'z' を 0～25 のインデックスに対応付ける
			const int index = (lower - 'a');

			// 出現回数をインクリメント
			++counts[index];
		}
	}

	for (int i = 0; i < 26; ++i)
	{
		printf("%c: %d\n", ('A' + i), counts[i]);
	}
}

    14.7.2¶

    ヒストグラムを表示するプログラム
    #include <stdio.h>
#include <stdlib.h>
#include <time.h>

// min 以上 max 以下のランダムな整数を返す関数
int RandomIntRange(int min, int max)
{
	return (rand() % (max - min + 1)) + min;
}

// 6 個のサイコロを振り、出目の合計（6～36）を返す関数
int Roll6D6()
{
	int sum = 0;

	for (int i = 0; i < 6; ++i)
	{
		sum += RandomIntRange(1, 6);
	}

	return sum;
}

int main()
{
	srand((unsigned int)time(NULL));

	const int n = 10000; // サイコロを振る回数

	// 各結果の出現回数を記録する配列
	// counts[score] が得点 score の出現回数を記録する
	int counts[37] = { };

	for (int i = 0; i < n; ++i)
	{
		const int score = Roll6D6(); // 6 以上 36 以下
		++counts[score];
	}

	// 表示が横に長くなりすぎないよう、何回ごとに '*' を 1 個表示するかを決める
	const int countsPerBlock = (n < 400) ? 1 : (n / 400);

	for (int score = 6; score <= 36; ++score)
	{
		printf("%2d: ", score);

		const int blocks = counts[score] / countsPerBlock;

		for (int k = 0; k < blocks; ++k)
		{
			putchar('*');
		}

		printf("(%d)\n", counts[score]);
	}
}
