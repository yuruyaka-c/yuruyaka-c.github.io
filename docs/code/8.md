### 8.1.2

```c title="n を target に近づけていくプログラム"
#include <stdio.h>

int main()
{
	int n, target;
	scanf("%d %d", &n, &target);

	while (n != target)
	{
		printf("%d\n", n);

		if (n < target)
		{
			++n;
		}
		else
		{
			--n;
		}
	}

	printf("最終値: %d\n", n);
}
```

### 8.4.1

```c title="条件演算子の使用例"
#include <stdio.h>

int main()
{
	// 条件式が真(1)なので 100 が選ばれる
	printf("%d\n", (1 ? 100 : 200));

	// 条件式が偽(0)なので 200 が選ばれる
	printf("%d\n", (0 ? 100 : 200));
}
```

### 8.4.2

```c title="条件演算子を使って、小さいほうの値、大きいほうの値を選ぶ"
#include <stdio.h>

int main()
{
	int a = 100, b = 200;

	// a < b のとき a、それ以外のとき b
	int min = (a < b) ? a : b;

	// a > b のとき a、それ以外のとき b
	int max = (a > b) ? a : b;

	printf("小さいほうの値: %d\n", min);
	printf("大きいほうの値: %d\n", max);
}
```

## 8.5

### 8.5.1

```c title="論理否定演算子の例"
#include <stdio.h>

int main()
{
	printf("%d\n", !1); // 1 は真なので、!1 は 0
	printf("%d\n", !0); // 0 は偽なので、!0 は 1

	printf("%d\n", !100); // 100 は真（非0）なので、!100 は 0
	printf("%d\n", !-5); // -5 は真（非0）なので、!-5 は 0

	printf("%d\n", (10 < 20)); // (10 < 20) は真なので 1
	printf("%d\n", !(10 < 20)); // !(10 < 20) は !1 なので 0
}
```

### 8.5.5

```c title="真偽の反転の例"
#include <stdio.h>

int main()
{
	int darkMode = 1;
	printf("ダークモード: %d\n", darkMode);

	// ダークモードの状態を反転して代入
	darkMode = !darkMode;
	printf("ダークモード: %d\n", darkMode);

	// ダークモードの状態を再度反転して代入
	darkMode = !darkMode;
	printf("ダークモード: %d\n", darkMode);
}
```
