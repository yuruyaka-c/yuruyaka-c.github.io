### 7.2.5

```c title="入力した文字列の大文字・小文字を反転するコード"
#include <stdio.h>
#include <ctype.h>

int main()
{
	for (;;)
	{
		int c = getchar();

		// 改行文字が入力されたらループを抜ける
		if (c == '\n')
		{
			break;
		}

		if (islower(c))
		{
			putchar(toupper(c)); // 小文字なら大文字に変換して出力
		}
		else if (isupper(c))
		{
			putchar(tolower(c)); // 大文字なら小文字に変換して出力
		}
		else
		{
			putchar(c); // それ以外の文字はそのまま出力
		}
	}

	putchar('\n');
}
```

```c title="入力した文字列に含まれる数字をすべて足し合わせるコード"
#include <stdio.h>
#include <ctype.h>

int main()
{
	int sum = 0;

	{

		// 改行文字が入力されたらループを抜ける
		if (c == '\n')
		{
			break;
		}

		if (isdigit(c))

			// 数字なら整数に変換して加算
			sum += (c - '0');
		}

}
```
