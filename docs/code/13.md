# 第 13 章 時間と乱数

## 13.1

### 13.1.1

```c title="time 関数の使用例"
#include <stdio.h>
#include <time.h>

int main()
{
	// 現在時刻（UNIX 時間）を取得
	const time_t currentTime = time(NULL);

	// long long 型にキャストして出力
	printf("現在時刻（UNIX 時間）: %lld 秒\n", (long long)currentTime);
}
```

### 13.1.2

```c title="現在時刻を利用した値の生成例"
#include <stdio.h>
#include <time.h>

int main()
{
	// 現在時刻を取得
	const time_t currentTime = time(NULL);

	// 0 から 3 までの値を生成
	const int value = (int)((long long)currentTime % 4);

	printf("生成された値: %d\n", value);
}
```


### 13.1.3

```c title="擬似乱数のしくみを理解するための簡易的な例"
#include <stdio.h>

// 説明のための簡易的な擬似乱数生成関数。0～32767 の範囲の整数を生成
int SimpleRand(int previous)
{
	return (int)((previous * 1103515245LL + 12345) % 32768);
}

int main()
{
	const int seed = 1; // シードを 1 に設定
	const int a = SimpleRand(seed); // seed を使って最初の乱数を生成
	const int b = SimpleRand(a); // a を使って次の乱数を生成
	const int c = SimpleRand(b); // b を使って次の乱数を生成

	printf("%d\n", a);
	printf("%d\n", b);
	printf("%d\n", c);
}
```

## 13.3

### 13.3.1

```c title="0.0 から 1.0 までの範囲の乱数を生成する関数"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

// [0.0, 1.0] の範囲の乱数を返す関数
double RandomDouble()
{
	return (double)rand() / RAND_MAX;
}

int main()
{
	srand((unsigned int)time(NULL));

	// 5 個の乱数を生成して出力

		printf("%f\n", RandomDouble());
```

### 13.3.3

```c title="0 から max までの範囲の整数乱数を生成する関数"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

// [0, max] の範囲の整数乱数を返す関数
int RandomInt(int max)

	// 5 個の整数乱数を生成して出力

	}
```

### 13.3.4

```c title="min から max までの範囲の整数乱数を生成する関数"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

// [min, max] の範囲の整数乱数を返す関数
int RandomIntRange(int min, int max)
{
	return (rand() % (max - min + 1)) + min;
}

int main()
{
	srand((unsigned int)time(NULL));

	// 10 個の整数乱数を生成して出力
	for (int i = 0; i < 10; ++i)
	{
		printf("%d\n", RandomIntRange(1, 6)); // 1 ～ 6 の乱数
	}
}
```

```c title="おみくじプログラム"
int main()
{
	srand((unsigned int)time(NULL));
	const int result = rand() % 4; // 0 ～ 3 の整数乱数を生成

		break;
	case 1:
		puts("中吉");
		break;
	case 2:
		puts("小吉");
		break;
	case 3:
		puts("凶");
		break;
	}
}
```

### 13.4.2

```c title="確率にバイアスのあるおみくじプログラム"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

// [0.0, 1.0) の範囲の乱数を返す関数
double RandomDoubleOpen()
{
	return (double)rand() / ((double)RAND_MAX + 1.0);
}

int main()
{
	srand((unsigned int)time(NULL));
	const double result = RandomDoubleOpen();
	printf("おみくじの結果: ");

	if (result < 0.35) // [0.0, 0.35)
	{
		puts("大吉"); // 35% の確率
	}
	else if (result < 0.70) // [0.35, 0.70)
	{
		puts("中吉"); // 35% の確率
	}
	else if (result < 0.90) // [0.70, 0.90)
	{
		puts("小吉"); // 20% の確率
	}
	else // [0.90, 1.0)
	{
		puts("凶"); // 10% の確率
	}
}
```

### 13.4.3

```c title="モンテカルロ法による円周率の近似"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

// [0.0, 1.0] の範囲の乱数を返す関数
double RandomDouble()
{
	return (double)rand() / RAND_MAX;
}

int main()
{
	srand((unsigned int)time(NULL));

	printf("試行回数を入力してください > ");
	int n;
	scanf("%d", &n);

	// 扇形の内側に入った点の個数
	int insideSector = 0;

	for (int i = 0; i < n; ++i)
	{
		const double x = RandomDouble(); // x 座標
		const double y = RandomDouble(); // y 座標

		// 原点からの距離の 2 乗が 1.0 以下なら扇形の内側
		if ((x * x + y * y) <= 1.0)
		{
			++insideSector;
		}
	}

	const double result = (double)insideSector / n * 4.0;
	printf("近似された円周率: %f\n", result);
}
```
