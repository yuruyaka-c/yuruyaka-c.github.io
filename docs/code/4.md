## 4.1

### 4.1.3

```c title="比較式の結果を確認するコード"
#include <stdio.h>

int main()
{
	printf("5 < 10  : %d\n", (5 < 10));  // 1（真）
	printf("10 <= 5 : %d\n", (10 <= 5)); // 0（偽）
	printf("3 > 0   : %d\n", (3 > 0));   // 1（真）
	printf("3 >= 3  : %d\n", (3 >= 3));  // 1（真）
	printf("7 == 7  : %d\n", (7 == 7));  // 1（真）
	printf("7 == 8  : %d\n", (7 == 8));  // 0（偽）
	printf("7 != 8  : %d\n", (7 != 8));  // 1（真）
	printf("7 != 7  : %d\n", (7 != 7));  // 0（偽）
}
```

### 4.1.8

```c title="❌ 誤り: 連続比較は意図した結果にならない"
#include <stdio.h>

int main()
{
	printf("5 < 10 < 7   : %d\n", (5 < 10 < 7));   // 1（真）→ 期待と異なる
	printf("-10 < -5 < 0 : %d\n", (-10 < -5 < 0)); // 0（偽）→ 期待と異なる
}
```

```c title="（参考）連続比較の正しい書き方"
#include <stdio.h>

int main()
{
	printf("(5 < 10) && (10 < 7)  : %d\n", ((5 < 10) && (10 < 7)));   // 0（偽）
	printf("(-10 < -5) && (-5 < 0): %d\n", ((-10 < -5) && (-5 < 0))); // 1（真）
}
```

### 4.5.4

```c title="BMI 判定のコード"
#include <stdio.h>

int main()
{
	printf("身長（cm）を入力してください > ");
	double cm;
	scanf("%lf", &cm);

	printf("体重（kg）を入力してください > ");
	double kg;
	scanf("%lf", &kg);

	// 身長をメートルに変換する
	double m = cm / 100.0;

	// BMI を計算する
	double bmi = kg / (m * m);

	if (bmi < 18.5)
	{
		puts("低体重");
	}
	else if (bmi < 25.0)
	{
		puts("標準");
	}
	else if (bmi < 30.0)
	{
		puts("太り気味");
	}
	else
	{
		puts("肥満");
	}
}
```
