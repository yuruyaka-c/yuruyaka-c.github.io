    第 12 章 値渡しとポインタ渡し¶

    12.1¶

    12.1.1¶

    ❌ 期待どおりに動かない例
    #include <stdio.h>

// 回復ドリンクを使い、HP を回復させる関数
void UsePotion(int hp, int count)
{
	if (0 < count)
	{
		--count; // ドリンクを 1 個消費
		hp += 50; // HP を 50 回復
		printf("【関数内】回復した。HP: %d, ドリンク残数: %d\n", hp, count);
	}
	else
	{
		puts("【関数内】ドリンクがない！");
	}
}

int main()
{
	int playerHP = 50; // プレイヤーの HP
	int potionCount = 5; // 回復ドリンクの所持数

	printf("【関数呼び出し前】HP: %d, ドリンク残数: %d\n", playerHP, potionCount);
	UsePotion(playerHP, potionCount); // 回復ドリンクを使う
	printf("【関数呼び出し後】HP: %d, ドリンク残数: %d\n", playerHP, potionCount);
}

    12.2¶

    12.2.1¶

    アドレス演算子（&）を使って変数のアドレスを取得する例
    #include <stdio.h>

int main()
{
	int value = 100;
	printf("value の値: %d\n", value);
	printf("value のアドレス: %p\n", &value);
}

}

    12.2.3¶

    ポインタ変数の使用例
    #include <stdio.h>

int main()
{
	int age = 20;
	int* pAge = &age; // ポインタ変数 pAge に age のアドレスを代入する

	printf("&age: %p\n", &age); // age のアドレスを出力する
	printf("pAge: %p\n", pAge); // pAge の値（age のアドレス）を出力する
}

    12.2.4¶

    ポインタ変数を使った値の変更例
    #include <stdio.h>

int main()
{
	int age = 20;
	int* pAge = &age; // ポインタ変数 pAge に age のアドレスを代入する

	printf("変更前: age = %d\n", age);
	printf("pAge が指すアドレスにある値 = %d\n", *pAge); // pAge が指すアドレスにある値（age の値）を出力する

	*pAge = 30; // pAge が指すアドレスにある値（age の値）を 30 に変更する

	printf("変更後: age = %d\n", age);
	printf("pAge が指すアドレスにある値 = %d\n", *pAge); // pAge が指すアドレスにある値（age の値）を出力する
}

    12.3.2¶

    ❌ 期待どおりに動かない例
    #include <stdio.h>

// 2 つの整数の値を交換するつもりの関数
void SwapInt(int* a, int* b)
{
	printf("【関数内（交換前）】*a = %d, *b = %d\n", *a, *b);

	int* t = a; // ポインタ変数 t に a の値（アドレス）を退避
	a = b;      // a に b の値（アドレス）を代入
	b = t;      // b に退避しておいた値（アドレス）を代入

	printf("【関数内（交換後）】*a = %d, *b = %d\n", *a, *b);
}

int main()
{
	int x = 10;
	int y = 20;

	printf("【関数呼び出し前】x = %d, y = %d\n", x, y);
	SwapInt(&x, &y);
	printf("【関数呼び出し後】x = %d, y = %d\n", x, y);
}
    ⭕️ 正しい値の交換の例
    #include <stdio.h>

// 2 つの整数の値を交換する関数
void SwapInt(int* a, int* b)
{
	printf("【関数内（交換前）】*a = %d, *b = %d\n", *a, *b);

	int t = *a; // a が指す先の値を退避
	*a = *b;    // a が指す先に、 b が指す先の値を代入
	*b = t;     // b が指す先に、退避しておいた値を代入

	printf("【関数内（交換後）】*a = %d, *b = %d\n", *a, *b);
}

int main()
{
	int x = 10;
	int y = 20;

	printf("【関数呼び出し前】x = %d, y = %d\n", x, y);
	SwapInt(&x, &y);
	printf("【関数呼び出し後】x = %d, y = %d\n", x, y);
}
