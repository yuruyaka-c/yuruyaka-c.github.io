    第 4 章 比較と条件分岐¶

    4.1¶

    4.1.3¶

    比較式の結果を確認するコード
    #include <stdio.h>

int main()
{
	printf("5 < 10  : %d\n", (5 < 10));  // 1（真）
	printf("10 <= 5 : %d\n", (10 <= 5)); // 0（偽）
	printf("3 > 0   : %d\n", (3 > 0));   // 1（真）
	printf("3 >= 3  : %d\n", (3 >= 3));  // 1（真）
	printf("7 == 7  : %d\n", (7 == 7));  // 1（真）
	printf("7 == 8  : %d\n", (7 == 8));  // 0（偽）
	printf("7 != 8  : %d\n", (7 != 8));  // 1（真）
	printf("7 != 7  : %d\n", (7 != 7));  // 0（偽）
}

    4.1.4¶

    浮動小数点数どうしの比較
    #include <stdio.h>

int main()
{
	printf("3.14 < 2.71 : %d\n", (3.14 < 2.71)); // 0（偽）
	printf("1.0 == 1.0  : %d\n", (1.0 == 1.0));   // 1（真）
}

    4.1.5¶

    整数と浮動小数点数の比較例
    #include <stdio.h>

int main()
{
	printf("1.5 < 2  : %d\n", (1.5 < 2));   // 1（真）
	printf("3 >= 3.0 : %d\n", (3 >= 3.0)); // 1（真）
	printf("4 == 4.1 : %d\n", (4 == 4.1)); // 0（偽）
}

    #include <stdio.h>

int main()
{
	double w1, w2, w3;
	scanf("%lf %lf %lf", &w1, &w2, &w3);
	printf("預け入れ可否: %d\n", ((w1 + w2 + w3) <= 20.0));

    4.1.8¶

    ❌ 誤り: 連続比較は意図した結果にならない
    #include <stdio.h>

int main()
{
	printf("5 < 10 < 7   : %d\n", (5 < 10 < 7));   // 1（真）→ 期待と異なる
	printf("-10 < -5 < 0 : %d\n", (-10 < -5 < 0)); // 0（偽）→ 期待と異なる
}
    （参考）連続比較の正しい書き方
    #include <stdio.h>

int main()
{
	printf("(5 < 10) && (10 < 7)  : %d\n", ((5 < 10) && (10 < 7)));   // 0（偽）
	printf("(-10 < -5) && (-5 < 0): %d\n", ((-10 < -5) && (-5 < 0))); // 1（真）
}

int main()

    4.5.3¶

    令和の日付判定のコード
    #include <stdio.h>

int main()
{
	int year, month, day;
	scanf("%d %d %d", &year, &month, &day);

	// 「2019 より大きい年」または「2019 年かつ 5 月以降」

	if ((2019 < year) || ((year == 2019) && (5 <= month)))
	{
		puts("令和");
	}
	else
	{

}
    令和の日付判定のコード（別解）
    #include <stdio.h>

int main()
{
	int year, month, day;
	scanf("%d %d %d", &year, &month, &day);

	// 日付を 1 つの整数値に変換する
	int date = (year * 10000 + month * 100 + day);

	if (20190501 <= date)
	{
		puts("令和");
	}
	else
	{
		puts("令和ではない");
	}
}

    4.5.4¶

    BMI 判定のコード
    #include <stdio.h>

int main()
{
	printf("身長（cm）を入力してください > ");
	double cm;
	scanf("%lf", &cm);

	printf("体重（kg）を入力してください > ");
	double kg;
	scanf("%lf", &kg);

	// 身長をメートルに変換する
	double m = cm / 100.0;

	// BMI を計算する
	double bmi = kg / (m * m);

	if (bmi < 18.5)
	{
		puts("低体重");
	}
	else if (bmi < 25.0)
	{
		puts("標準");
	}
	else if (bmi < 30.0)
	{
		puts("太り気味");
	}
	else
	{
		puts("肥満");
	}
}
