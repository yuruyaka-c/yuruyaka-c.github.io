    7.2.5¶

    入力した文字列の大文字・小文字を反転するコード
    #include <stdio.h>
#include <ctype.h>

int main()
{
	for (;;)
	{
		int c = getchar();

		// 改行文字が入力されたらループを抜ける
		if (c == '\n')
		{
			break;
		}

		if (islower(c))
		{
			putchar(toupper(c)); // 小文字なら大文字に変換して出力
		}
		else if (isupper(c))
		{
			putchar(tolower(c)); // 大文字なら小文字に変換して出力
		}
		else
		{
			putchar(c); // それ以外の文字はそのまま出力
		}
	}

	putchar('\n');
}

    7.2.6¶

    入力した文字列に含まれる数字をすべて足し合わせるコード
    #include <stdio.h>
#include <ctype.h>

int main()
{
	int sum = 0;

	for (;;)
	{
		int c = getchar();

		// 改行文字が入力されたらループを抜ける
		if (c == '\n')
		{
			break;
		}

		if (isdigit(c))
		{
			// 数字なら整数に変換して加算
			sum += (c - '0');
		}
	}

	printf("合計: %d\n", sum);
}

    素数判定のプログラム
    #include <stdio.h>

int main()
{
	printf("整数 n を入力 > ");
	int n;
	scanf("%d", &n);

	if (n <= 1)
	{
		puts("素数ではない");
		return 0;
	}

	for (int i = 2; i <= (n / 2); ++i)
	{
		if ((n % i) == 0) // 割り切れた場合

			return 0;
		}
	}

}

    7.8.3¶

    二重ループにおける break の例
    #include <stdio.h>

int main()
{
	for (int year = 2020; year <= 2025; ++year)
	{
		for (int month = 1; month <= 12; ++month)
		{
			if ((year == 2020) && (month == 3))
