# 第 17 章 構造体

## 17.1

```c title="2 つの日付を個別の変数で扱う例"
#include <stdio.h>

void PrintDate(int year, int month, int day)
{
	printf("%04d-%02d-%02d\n", year, month, day);
}

int main()
{
	int year1 = 2027;
	int month1 = 1;
	int day1 = 1;

	int year2 = 2027;
	int month2 = 12;
	int day2 = 31;

	PrintDate(year1, month1, day1);
	PrintDate(year2, month2, day2);
}
```

```c title="構造体を使って日付を管理する例"
#include <stdio.h>

// 日付を表す構造体 Date を定義
struct Date
{
	int year;
	int month;
	int day;
};

void PrintDate(struct Date date)
{
	printf("%04d-%02d-%02d\n", date.year, date.month, date.day);
}

int main()
{
	struct Date date1 = { 2027, 1, 1 };
	struct Date date2 = { 2027, 12, 31 };

	PrintDate(date1);
	PrintDate(date2);
}
```

### 17.2.2

```c title="構造体のポインタ渡しの例"
#include <stdio.h>

typedef struct
{
	int x;
	int y;
} Point;

void PrintPoint(const Point* p)
{
	printf("(%d, %d)\n", (*p).x, (*p).y);
}

void ChangePoint(Point* p)
{
	// 呼び出し元の構造体のメンバを変更できる
	(*p).x += 100;
	(*p).y += 200;
}

int main()
{
	Point point = { 10, 20 };
	PrintPoint(&point);
	ChangePoint(&point);
	PrintPoint(&point);
}
```

### 17.2.3

```c title="関数から構造体を返す例"
#include <stdio.h>

typedef struct
{
	int x;
	int y;
} Point;

Point CreatePoint(int x, int y)
{
	Point p = { x, y };
	return p;
}

Point AddPoints(Point p1, Point p2)
{
	Point result = { (p1.x + p2.x), (p1.y + p2.y) };
	return result;
}

void PrintPoint(Point p)
{
	printf("(%d, %d)\n", p.x, p.y);
}

int main()
{
	Point p1 = CreatePoint(10, 20);
	Point p2 = CreatePoint(30, 40);
	Point p3 = AddPoints(p1, p2);

	PrintPoint(p1);
	PrintPoint(p2);
	PrintPoint(p3);
}
```

## 17.3

### 17.3.1

```c title="配列をメンバに持つ構造体の例"
#include <stdio.h>

typedef struct
{
	int id; // 学生番号
	int scores[3]; // 点数（3 科目分）
} Student;

void PrintStudent(const Student* s)
{
	printf("学生番号: %d\n", s->id);
	printf("点数: ");

	for (int i = 0; i < 3; ++i)
	{
		printf("%d ", s->scores[i]);
	}

	putchar('\n');
}

int main()
{
	// 配列メンバの初期化は { ... } の中に入れ子で記述する
	Student student1 = { 1001, { 85, 90, 78 } };
	Student student2 = { 1002, { 92, 88, 95 } };

	PrintStudent(&student1);
	PrintStudent(&student2);

	student1 = student2; // 配列メンバも含めて構造体全体を代入できる
	PrintStudent(&student1);
}
```

```c title="構造体の比較例"
#include <stdio.h>

typedef struct
{
	int x;
	int y;
} Point;

// 2 つの Point が等しいかどうかを返す関数
bool PointEqual(Point p1, Point p2)
{
	return (p1.x == p2.x) && (p1.y == p2.y);
}

int main()
{
	Point p1 = { 10, 20 };
	Point p2 = { 10, 20 };
	Point p3 = { 30, 40 };

	printf("p1 と p2 は等しい: %s\n", (PointEqual(p1, p2) ? "true" : "false"));
	printf("p1 と p3 は等しい: %s\n", (PointEqual(p1, p3) ? "true" : "false"));
}
```

### 17.3.4

```c title="複合リテラルの例"
#include <stdio.h>

typedef struct
{
	int x;
	int y;
} Point;

void PrintPoint(Point p)
{
	printf("(%d, %d)\n", p.x, p.y);

	return (Point){ (p1.x + p2.x), (p1.y + p2.y) };

int main()

	PrintPoint(p2);
```

### 17.3.5

```c title="構造体の配列の例①"
#include <stdio.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

typedef struct
{
	char name[51];
	int id;
	int score;
} Student;

int main()
{
	// Student 型の要素を 3 つ持つ配列を定義・初期化
	Student students[3] =
	{
		{ "Alice", 1001, 85 },
		{ "Bob", 1002, 92 },
		{ "Carol", 1003, 78 }
	};

	// 配列の各要素にアクセスしてメンバを出力
	for (int i = 0; i < COUNTOF(students); ++i)
	{
		puts("--------");
		printf("名前: %s\n", students[i].name);
		printf("学生番号: %d\n", students[i].id);
		printf("点数: %d\n", students[i].score);
	}
}
```
