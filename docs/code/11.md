# 第 11 章 const とマクロ

## 11.1

### 11.1.1

```c title="❌ const を使わない場合に起こる問題の例"
#include <stdio.h>

int main()
{
	double pi = 3.1415926535; // 円周率
	pi = pi * 2.0; // ミス: 円周率の値を上書きしてしまった
	printf("円周率: %f\n", pi);
}
```

### 11.1.2

```c title="⚠️ const を付けた変数の値の変更"
#include <stdio.h>

int main()
{
	const double pi = 3.1415926535; // 円周率（読み取り専用）
	pi = pi * 2.0; // コンパイルエラー: const 変数に代入しようとした
	printf("円周率: %f\n", pi);
}
```

```c title="❌ 誤り: const 変数に初期値を与えていない"
#include <stdio.h>

int main()
{
	const int maxUser;
	maxUser = 100; // コンパイルエラー
	printf("最大ユーザー数: %d\n", maxUser);
}
```

```c title="❌ 誤り: const 変数の値を変更しようとしている"
#include <stdio.h>

int main()
{
	const int maxUser = 100;
	maxUser = 200; // コンパイルエラー
	maxUser += 50; // コンパイルエラー
	++maxUser;     // コンパイルエラー
	scanf("%d", &maxUser); // コンパイルエラー
	printf("最大ユーザー数: %d\n", maxUser);
}
```

### 11.1.3

```c title="⚠️ マジックナンバーが登場する例"
#include <stdio.h>

int main()
{
	int price = 1000;
	int priceWithTax = (int)(price * 1.1); // 1.1 がマジックナンバー
	printf("税込価格: %d 円\n", priceWithTax);
}
```

```c title="⭕️ マジックナンバーを const 変数で置き換えた例"
#include <stdio.h>

int main()
{
	const double taxRate = 0.1; // 消費税率 10%
	int price = 1000;
	int priceWithTax = (int)(price * (1.0 + taxRate)); // マジックナンバーを使わない
	printf("税込価格: %d 円\n", priceWithTax);
}
```

## 11.2

### 11.2.2

```c title="マクロ定数の使用例"
#include <stdio.h>

#define PI 3.141592653589793
#define MAX_SCORE 100
#define GAME_TITLE "C Programming Adventure"

// 円の面積を計算する関数
double GetCircleArea(double radius)
{
	return PI * radius * radius;
}

// 円の円周を計算する関数
double GetCircleCircumference(double radius)
{
	return 2.0 * PI * radius;
}

int main()
{
	puts(GAME_TITLE);
	printf("最大スコア: %d\n", MAX_SCORE);

	double radius = 5.0;
	printf("半径 %.2f の円の面積: %.2f\n", radius, GetCircleArea(radius));
	printf("半径 %.2f の円の円周: %.2f\n", radius, GetCircleCircumference(radius));
}
```

### 11.2.3

```c title="誤り：case ラベルにconst 変数を使っている"
#include <stdio.h>

int main()
{
	const int OPTION_A = 1;
	const int OPTION_B = 2;

	int option = OPTION_A;

	switch (option)
	{
	case OPTION_A: // コンパイルエラー
		puts("Option A selected.");
		break;
	case OPTION_B: // コンパイルエラー
		puts("Option B selected.");
		break;
	}
}
```

```c title="case ラベルにマクロ定数を使う場合"
#include <stdio.h>

#define OPTION_A 1
#define OPTION_B 2

int main()
{
	int option = OPTION_A;

	switch (option)
	{
	case OPTION_A:
		puts("Option A selected.");
		break;
	case OPTION_B:
		puts("Option B selected.");
		break;
	}
}
```

```c title="配列の要素数の定義例"
#include <stdio.h>

#define PLAYERS_COUNT 4

int main()
{
	int scores[PLAYERS_COUNT] = { 0 }; // 配列の要素数にマクロ定数を使う

	for (int i = 0; i < PLAYERS_COUNT; ++i)
	{
		printf("Player %d score: %d\n", (i + 1), scores[i]);
	}
}
```

## 11.3

### 11.3.1

```c title="関数形式マクロの使用例"
#include <stdio.h>

#define SQUARE(x) ((x) * (x))

int main()
{
	int a = SQUARE(5);
	int b = SQUARE(a);

	printf("SQUARE(5) = %d\n", a);
	printf("SQUARE(a) = %d\n", b);
	printf("SQUARE(1.5) = %.2f\n", SQUARE(1.5));
}
```

```c title="MAX マクロの定義例"
#include <stdio.h>

#define MAX(a, b) ((a) > (b) ? (a) : (b))

int main()
{
	printf("MAX(10, 20) = %d\n", MAX(10, 20));
	printf("MAX(3.14, 2.71) = %.2f\n", MAX(3.14, 2.71));
}
```

### 11.3.2

```c title="❌ 誤り: 引数をかっこで囲んでいない"
#include <stdio.h>

#define SQUARE(x) (x * x)

int main()
{
	int a = SQUARE(1 + 2); // (1 + 2 * 1 + 2) に展開されてしまう
	printf("%d\n", a);
}
```

```c title="⭕️ 引数をかっこで囲む場合の例"
#include <stdio.h>

#define SQUARE(x) ((x) * (x))

int main()
{
	int a = SQUARE(1 + 2); // ((1 + 2) * (1 + 2)) に展開される
	printf("%d\n", a);
}
```

```c title="❌ 誤り: マクロ全体をかっこで囲んでいない"
#include <stdio.h>

#define ADD(x, y) (x) + (y)

int main()
{
	int a = 10 * ADD(2, 3); // 10 * (2) + (3) に展開されてしまう
	printf("%d\n", a);
}
```

```c title="⭕️ マクロ全体をかっこで囲む場合の例"
#include <stdio.h>

#define ADD(x, y) ((x) + (y))

int main()
{
	int a = 10 * ADD(2, 3); // 10 * ((2) + (3)) に展開される
	printf("%d\n", a);
}
```

```c title="❌ 誤り: 副作用のある式を関数形式マクロの引数に使っている"
#include <stdio.h>

#define MAX(a, b) ((a) > (b) ? (a) : (b))

int main()
{
	int i = 0;
	int maxValue = MAX(++i, 0); // ++i が複数回評価されてしまう
	printf("maxValue = %d\n", maxValue);
	printf("i = %d\n", i);
}
```

```c title="⭕️ 通常の関数を使う場合の例"
#include <stdio.h>

// 2 つの整数のうち大きい方を返す関数
int Max(int a, int b)
{
	return (a > b) ? a : b;
}

int main()
{
	int i = 0;
	int maxValue = Max(++i, 0); // ++i は 1 回だけ評価される
	printf("maxValue = %d\n", maxValue);
	printf("i = %d\n", i);
}
```
