    第 1 章 プログラミングの基礎¶

    1.1¶

    目標 ① の完成形
    int main()
{

}
    目標 ② の完成形
    #include <stdio.h>

int main()
{
	puts("Apple");
}

    空っぽの main 関数
    int main()
{

}

    1.1.4¶

    <stdio.h> をインクルードするコード
    #include <stdio.h>

int main()
{

}

    1.1.5¶

    puts 関数を使うコード
    #include <stdio.h>

int main()
{
	puts("Apple");
}

    1.1.6¶

    puts 関数の前にあるインデント
    #include <stdio.h>

int main()
{
	puts("Apple");
}

    1.1.7¶

    正常終了の報告を明示的に行うコード
    #include <stdio.h>

int main()
{
	puts("Apple");
	return 0;
}

    1.1.8¶

    正常終了の報告を自動的に行うコード
    #include <stdio.h>

int main()
{
	puts("Apple");
}

int main()

	puts("C Language");

    #include <stdio.h>

int main()
{
	puts("リンゴが木から落ちた");
}

int main()

    1.2¶

    1.2.1¶

    複数の puts 関数を使うコード
    #include <stdio.h>

int main()
{
	puts("Apple");
	puts("Banana");
}

    1.2.2¶

    puts 関数の並びを入れ替えたコード
    #include <stdio.h>

int main()
{
	puts("Banana");
	puts("Apple");
}

    1.2.3¶

    puts 関数を増やしたコード
    #include <stdio.h>

int main()
{
	puts("Apple");
	puts("Banana");
	puts("Cat");
}

    1.2.4¶

    インデントが不ぞろいなコード
    #include <stdio.h>

int main()
{
	puts("Apple");
 puts("Banana");
  puts("Cat");
puts("Dog");
}
    インデントがそろっているコード
    #include <stdio.h>

int main()
{
	puts("Apple");
	puts("Banana");
	puts("Cat");
	puts("Dog");
}

    1.2.5¶

    文字を並べて長方形を描くコード
    #include <stdio.h>

int main()
{
	puts("**********");
	puts("*        *");
	puts("*        *");
	puts("*        *");
	puts("**********");
}
    文字を並べて三角形を描くコード
    #include <stdio.h>

int main()
{
	puts("    *");
	puts("   ***");
	puts("  *****");
	puts(" *******");
	puts("*********");
}

    1.2.6¶

    文字列の中で改行文字を使うコード
    #include <stdio.h>

int main()
{
	puts("Apple\nBanana\nCat");
	puts("Dog");
}

    1.2.7¶

    たくさん改行するコード
    #include <stdio.h>

int main()
{
	puts("Apple\n\nBanana\n\n\nCat\n");
	puts("Dog");
}

    1.3¶

    1.3.1¶

    文字列の中で改行文字を使うコード
    #include <stdio.h>

int main()
{
	puts("He waited.\n\nNo one came.");
}

    1.3.2¶

    文字列の中でタブ空白を使うコード
    #include <stdio.h>

int main()
{
	puts("1\tJanuary");
	puts("2\tFebruary");
	puts("3\tMarch");
	puts("12\tDecember");
}

    1.3.3¶

    文字列の中でダブルクォーテーションを使うコード
    #include <stdio.h>

int main()
{
	puts("She found the \"key\"!");
}

    1.3.4¶

    バックスラッシュを出力するコード
    #include <stdio.h>

int main()
{
	puts("/(^o^)\\");
}

    1.3.5¶

    エスケープシーケンスの練習
    #include <stdio.h>

int main()
{
	puts("/(^o^)\\n\\(^o^)/");
}
    タブ空白で整列された表を出力するコード
    #include <stdio.h>

int main()
{
	puts("|0\t|0\t|");
	puts("|1\t|1\t|");
	puts("|10\t|2\t|");
	puts("|11\t|3\t|");
}

int main()
{
	puts(Apple);
}

    1.4.3¶

    ❌ 誤り: main 関数の閉じかっこ } がないコード

int main()
{
	puts("Apple");

    1.4.4¶

    #include <stdio.h>

int main()

	puts("Apple\mBanana");
