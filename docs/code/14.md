### 14.7.1

```c title="アルファベットの出現回数を数えるプログラム"
#include <stdio.h>
#include <ctype.h>

int main()
{
	// 各アルファベットの出現回数を格納する配列
	// counts[0] が 'A'/'a' の出現回数
	// counts[1] が 'B'/'b' の出現回数
	// ...
	// counts[25] が 'Z'/'z' の出現回数
	int counts[26] = { };

	for (;;)
	{
		const int c = getchar();

		// 改行が来たら終了する
		if (c == '\n')
		{
			break;
		}

		if (isalpha(c))
		{
			// 小文字にそろえる
			const int lower = tolower(c);

			// 'a'～'z' を 0～25 のインデックスに対応付ける
			const int index = (lower - 'a');

			// 出現回数をインクリメント
			++counts[index];
		}
	}

	for (int i = 0; i < 26; ++i)
	{
		printf("%c: %d\n", ('A' + i), counts[i]);
	}
}
```

### 14.7.2

```c title="ヒストグラムを表示するプログラム"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

// min 以上 max 以下のランダムな整数を返す関数
int RandomIntRange(int min, int max)
{
	return (rand() % (max - min + 1)) + min;
}

// 6 個のサイコロを振り、出目の合計（6～36）を返す関数
int Roll6D6()
{
	int sum = 0;

	for (int i = 0; i < 6; ++i)
	{
		sum += RandomIntRange(1, 6);
	}

	return sum;
}

int main()
{
	srand((unsigned int)time(NULL));

	const int n = 10000; // サイコロを振る回数

	// 各結果の出現回数を記録する配列
	// counts[score] が得点 score の出現回数を記録する
	int counts[37] = { };

	for (int i = 0; i < n; ++i)
	{
		const int score = Roll6D6(); // 6 以上 36 以下
		++counts[score];
	}

	// 表示が横に長くなりすぎないよう、何回ごとに '*' を 1 個表示するかを決める
	const int countsPerBlock = (n < 400) ? 1 : (n / 400);

	for (int score = 6; score <= 36; ++score)
	{
		printf("%2d: ", score);

		const int blocks = counts[score] / countsPerBlock;

		for (int k = 0; k < blocks; ++k)
		{
			putchar('*');
		}

		printf("(%d)\n", counts[score]);
	}
}
```
