# 第 19 章 動的メモリ確保

## 19.1

### 19.1.1

```c title="⚠️ スタックオーバーフローを起こしやすいコードの例"
#include <stdio.h>
int main()
{
	// 要素数が 1000 万個の int 型配列（約 40 MB）
	// 実行環境によってはスタックオーバーフローになる可能性が高い
	int largeArray[10000000] = {};

	printf("%d\n", largeArray[0]);
}
```

```c title="⭕️ ヒープ領域を使って大きな配列を扱うコードの例"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	// 1000 万個の int 型配列用のメモリをヒープ領域に動的に確保
	int* largeArray = (int*)malloc(10000000 * sizeof(int));

	// メモリ確保に失敗した場合の処理
	if (largeArray == NULL)
	{
		puts("メモリの確保に失敗");
		return EXIT_FAILURE;
	}

	for (int i = 0; i < 10000000; ++i)
	{
		largeArray[i] = 0; // 配列の初期化
	}

	printf("%d\n", largeArray[0]);

	// 確保したメモリを解放
	free(largeArray);
}
```

## 19.2

### 19.2.3

```c title="int 型 1 個分のメモリを確保する例"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	// int 型 1 個分のメモリを動的に確保
	int* p = (int*)malloc(sizeof(int));

	// メモリ確保に失敗した場合の処理
	if (p == NULL)
	{
		puts("メモリの確保に失敗");
		return EXIT_FAILURE;
	}

	*p = 123; // 確保した領域に値を代入
	printf("%d\n", *p);

	*p += 100;
	printf("%d\n", *p);

	// 確保したメモリを解放
	free(p);
}
```

### 19.2.5

```c title="スタック領域とヒープ領域のアドレスの確認"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	// スタック領域にある変数
	int a = 10, b = 20;

	// ヒープ領域に確保したメモリ
	int* p = (int*)malloc(sizeof(int));

	if (p == NULL)
	{
		puts("メモリの確保に失敗");
		return EXIT_FAILURE;
	}

	printf("スタック領域の変数 a のアドレス: %p\n", &a);
	printf("スタック領域の変数 b のアドレス: %p\n", &b);
	printf("p が指すヒープ領域のアドレス: %p\n", p);

	free(p);
}
```

## 19.3

### 19.3.1

```c title="入力した要素数で配列を作る例"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	printf("配列の要素数を入力してください > ");
	int n;
	scanf("%d", &n);

	// int 型 n 個分のメモリを動的に確保
	int* array = (int*)malloc(n * sizeof(int));

	if (array == NULL)
	{
		puts("メモリの確保に失敗");
		return EXIT_FAILURE;
	}

	// 配列の各要素に値を代入
	for (int i = 0; i < n; ++i)
	{
		array[i] = (i * 10);
	}

	// 配列の各要素を出力
	for (int i = 0; i < n; ++i)
	{
		printf("array[%d] = %d\n", i, array[i]);
	}

	free(array);
}
```

## 19.4

### 19.4.1

```c title="構造体を動的に確保する例"
#include <stdio.h>
#include <stdlib.h>

typedef struct
{
	int x;
	int y;
} Point;

int main()
{
	// Point 型 1 個分のメモリを動的に確保
	Point* p = (Point*)malloc(sizeof(Point));

	if (p == NULL)
	{
		puts("メモリの確保に失敗");
		return EXIT_FAILURE;
	}

	// メンバに値を代入
	p->x = 10;
	p->y = 20;
	printf("(%d, %d)\n", p->x, p->y);

	free(p);
}
```

### 19.4.2

```c title="構造体配列を動的に確保する例"
#include <stdio.h>
#include <stdlib.h>

typedef struct
{
	int id;
	int score;
} Student;

int main()
{
	const int n = 3;

	// Student 型 n 個分のメモリを動的に確保
	Student* students = (Student*)malloc(n * sizeof(Student));

	if (students == NULL)
	{
		puts("メモリの確保に失敗");
		return EXIT_FAILURE;
	}

	// 値の代入
	for (int i = 0; i < n; ++i)
	{
		students[i].id = (1000 + i);
		students[i].score = (i * 10);
	}

	// 出力
	for (int i = 0; i < n; ++i)
	{
		printf("ID: %d, 点数: %d\n", students[i].id, students[i].score);
	}

	free(students);
}
```

```c title="ポインタをメンバに持つ構造体を動的に確保する例"
#include <stdio.h>
#include <stdlib.h>

typedef struct
{
	int id;
	int* scores;
} Student;

int main()
{
	// まず構造体本体を動的に確保
	Student* student = (Student*)malloc(sizeof(Student));

	if (student == NULL)
	{
		puts("メモリの確保に失敗");
		return EXIT_FAILURE;
	}

	student->id = 1001;

	// 実施した試験の回数
	const int testCount = 5;

	// 次に、scores が指す配列を動的に確保
	student->scores = (int*)malloc(testCount * sizeof(int));

	if (student->scores == NULL)
	{
		puts("メモリの確保に失敗");
		free(student); // 構造体本体を解放
		return EXIT_FAILURE;
	}

	// 試験の点数を代入
	for (int i = 0; i < testCount; ++i)
	{
		student->scores[i] = (i * 10);
	}

	// 学生の情報を表示
	printf("学生 ID: %d\n", student->id);

	for (int i = 0; i < testCount; ++i)
	{
		printf("試験 %d: %d 点\n", i, student->scores[i]);
	}

	// 先に配列を解放
	free(student->scores);
	// 最後に構造体本体を解放
	free(student);
}
```
