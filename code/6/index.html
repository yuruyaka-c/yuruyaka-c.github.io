    第 6 章 文字を扱う¶

    6.1¶

    6.1.1¶

    文字を出力する
    #include <stdio.h>

int main()
{
	printf("%c\n", 65);
	printf("%c\n", 66);
	printf("%c%c\n", 67, 100);
	printf("%c%c%c%c\n", 48, 49, 32, 33);
}

    6.1.2¶

    char 型の変数を使う例
    #include <stdio.h>

int main()
{
	char c1 = 65;
	char c2 = 98;
	char c3 = 53;
	printf("%c %c %c\n", c1, c2, c3);
}
    int 型の変数を使う例
    #include <stdio.h>

int main()
{
	int c1 = 65;
	int c2 = 98;
	int c3 = 53;
	printf("%c %c %c\n", c1, c2, c3);
}

    6.1.3¶

    文字リテラルの例
    #include <stdio.h>

int main()
{
	int c1 = 'A';
	printf("%c\n", c1);

	char c2 = 'B';
	printf("%c\n", c2);

	printf("%c\n", 'C');
	printf("%d\n", 'C');
}

    6.1.4¶

    特殊な文字の例
    #include <stdio.h>

int main()
{
	printf("Apple%cBanana%c", ' ', '\n');
	printf("A%cB%cC%c", '\t', '\t', '\n');
}

    6.2¶

    6.2.1¶

    文字の足し算
    #include <stdio.h>

int main()
{
	printf("%d\n", ('#' + 1));   // 35 + 1
	printf("%d\n", ('#' + '1')); // 35 + 49
	printf("%d\n", ('#' + 'A')); // 35 + 65
}
    文字の引き算
    #include <stdio.h>

int main()
{
	printf("%d\n", ('b' - 'a')); // 98 - 97
	printf("%d\n", ('z' - 1));   // 122 - 1
	printf("%d\n", ('5' - 5));   // 53 - 5
}
    文字の足し算・引き算の結果を数値と文字で出力するコード
    #include <stdio.h>

int main()
{
	printf("%d\n", ('A' + 1)); // 65 + 1
	printf("%d\n", ('A' + 2)); // 65 + 2
	printf("%d\n", ('Z' - 1)); // 90 - 1

	printf("%c\n", ('A' + 1)); // 65 + 1
	printf("%c\n", ('A' + 2)); // 65 + 2
	printf("%c\n", ('Z' - 1)); // 90 - 1
}

    6.2.2¶

    大文字・小文字の変換
    #include <stdio.h>

int main()
{
	printf("%c\n", ('A' + 32)); // 65 + 32
	printf("%c\n", ('R' + 32)); // 82 + 32

	printf("%c\n", ('a' - 32)); // 97 - 32
	printf("%c\n", ('r' - 32)); // 114 - 32
}

    6.2.3¶

    数字文字 → 整数値の変換
    #include <stdio.h>

int main()
{
	printf("%d\n", '0');
	printf("%d\n", '5');
	printf("%d\n", '9');

	printf("%d\n", ('0' - '0')); // 48 - 48
	printf("%d\n", ('5' - '0')); // 53 - 48
	printf("%d\n", ('9' - '0')); // 57 - 48
}

    6.2.4¶

    整数値 → 数字文字の変換
    #include <stdio.h>

int main()
{
	printf("%c\n", ('0' + 0)); // 48 + 0
	printf("%c\n", ('0' + 3)); // 48 + 3
	printf("%c\n", ('0' + 7)); // 48 + 7
}

    6.3¶

    6.3.1¶

    文字の入力
    #include <stdio.h>

int main()
{
	char c;
	scanf("%c", &c);
	printf("文字: %c\n", c);
	printf("整数値: %d\n", c);
}

    6.3.2¶

    複数の文字を入力する例
    #include <stdio.h>

int main()
{
	char c1, c2, c3;
	scanf("%c%c%c", &c1, &c2, &c3);
	printf("文字: %c %c %c\n", c1, c2, c3);
	printf("整数値: %d %d %d\n", c1, c2, c3);
}

    6.3.3¶

    数値入力は空白を読み飛ばす例
    #include <stdio.h>

int main()
{
	int a, b;
	scanf("%d %d", &a, &b);
	printf("整数値: %d %d\n", a, b);
}
    文字入力では空白文字を読み飛ばさない例
    #include <stdio.h>

int main()
{
	char c;
	scanf("%c", &c);
	printf("文字: %c\n", c);
	printf("整数値: %d\n", c);
}
    文字入力では改行文字を読み飛ばさない例
    #include <stdio.h>

int main()
{
	char c1, c2, c3;
	scanf("%c%c%c", &c1, &c2, &c3);
	printf("文字: %c %c %c\n", c1, c2, c3);
	printf("整数値: %d %d %d\n", c1, c2, c3);
}

    6.3.4¶

    文字入力で空白類文字を読み飛ばす例
    #include <stdio.h>

int main()
{
	char c;
	scanf(" %c", &c);
	printf("文字: %c\n", c);
	printf("整数値: %d\n", c);
}

    6.3.5¶

    読み飛ばしなしの例
    #include <stdio.h>

int main()
{
	char c1, c2, c3;
	scanf("%c%c%c", &c1, &c2, &c3);
	printf("%c(%d) %c(%d) %c(%d)\n", c1, c1, c2, c2, c3, c3);
}
    読み飛ばしありの例
    #include <stdio.h>

int main()
{
	char c1, c2, c3;
	scanf(" %c %c %c", &c1, &c2, &c3);
	printf("%c(%d) %c(%d) %c(%d)\n", c1, c1, c2, c2, c3, c3);
}

    6.3.6¶

    数値入力後の改行文字を読み込んでしまう例
    #include <stdio.h>

int main()
{
	int n;
	scanf("%d", &n);

	char c;
	scanf("%c", &c);

	printf("整数値: %d\n", n);
	printf("文字: %c(%d)\n", c, c);
}
    整数の直後の区切り文字を読み飛ばす例
    #include <stdio.h>

int main()
{
	int n;
	scanf("%d", &n);

	char c;
	scanf(" %c", &c);

	printf("整数値: %d\n", n);
	printf("文字: %c(%d)\n", c, c);
}

    6.4.1¶

    入力した文字が小文字であるかを調べる
    #include <stdio.h>

int main()
{
	char c;
	scanf("%c", &c);

	if (('a' <= c) && (c <= 'z'))
	{
		puts("小文字");
	}
	else
	{
		puts("小文字ではない");
	}
}

    6.4.2¶

    入力した文字が数字であるかを調べる
    #include <stdio.h>

int main()
{
	char c;
	scanf("%c", &c);

	if (('0' <= c) && (c <= '9'))
	{
		puts("数字");
	}
	else
	{
		puts("数字ではない");
	}
}
    ❌ 誤り: 数字文字と比較していない
    #include <stdio.h>

int main()
{
	char c;
	scanf("%c", &c);

	if ((0 <= c) && (c <= 9)) // 誤り
	{
		puts("数字");
	}
	else
	{
		puts("数字ではない");
	}
}

    6.4.3¶

    4 つの文字がパスワードと一致するかを調べる
    #include <stdio.h>

int main()
{
	char c1, c2, c3, c4;
	scanf("%c%c%c%c", &c1, &c2, &c3, &c4);

	if ((c1 == 'a') && (c2 == ' ') && (c3 == 'c') && (c4 == '!'))
	{
		puts("パスワードが一致");
	}
	else
	{
		puts("パスワードが不一致");
	}
}

    6.4.4¶

    簡単な数式パーサー（+ と -）
    #include <stdio.h>

int main()
{
	int a, b;
	char op; // operator（演算子）
	scanf("%d %c %d", &a, &op, &b);

	if (op == '+')
	{
		printf("%d\n", (a + b));
	}
	else if (op == '-')
	{
		printf("%d\n", (a - b));
	}
}
    簡単な数式パーサー（発展版）
    #include <stdio.h>

int main()
{
	double a, b;
	char op; // operator（演算子）
	scanf("%lf %c %lf", &a, &op, &b);

	if (op == '+')
	{
		printf("%f\n", (a + b));
	}
	else if (op == '-')
	{
		printf("%f\n", (a - b));
	}
	else if (op == '*')
	{
		printf("%f\n", (a * b));
	}
	else if (op == '/')
	{
		printf("%f\n", (a / b));
	}
}

    入力した文字がアルファベットであるかを判定する
    #include <stdio.h>
#include <ctype.h>

int main()
{
	int c = getchar();

    6.7¶

    6.7.1¶

    大文字に変換する・小文字に変換する
    #include <stdio.h>
#include <ctype.h>

int main()
{
	int c = getchar();
	int upper = toupper(c);
	int lower = tolower(c);
	printf("元の文字: %c\n", c);
	printf("大文字に変換した文字: %c\n", upper);
	printf("小文字に変換した文字: %c\n", lower);
}
