### 2.6.2

```c title="書式指定子を含まないフォーマット文字列の例"
{
	printf("App");
	printf("le\n");

```

### 2.6.3

```c title="フォーマット文字列のエスケープシーケンス %% を使うコード"
#include <stdio.h>

int main()
{
	printf("80%%\n");
	printf("%d%%\n", 90);

	// puts に渡す文字列はフォーマット文字列ではないため、
	// % の出力にエスケープシーケンスは不要
	puts("100%");
}
```

### 2.6.4

```c title="❌ 誤り: %% を使わなかった場合の事故の例"
#include <stdio.h>

int main()
{
	printf("50%dog! 100%frog!\n"); // %d や %f が書式指定子として解釈される
}
```

```c title="修正したコード"
#include <stdio.h>

int main()
{
	printf("50%%dog! 100%%frog!\n"); // %% を使って % を出力する
	puts("50%dog! 100%frog!"); // または puts を使う
}
```

### 2.7.3

```c title="ゼロ埋めを指定するコード"
#include <stdio.h>

int main()
{
	printf("%05d\n", 1);
	printf("%05d\n", 123);
	printf("%05d\n", 12345);

	// 時刻の出力に便利
	printf("%02d:%02d:%02d\n", 1, 2, 0);
	printf("%02d:%02d:%02d\n", 12, 23, 34);
}
```

### 2.7.4

```c title="左寄せをするコード"
#include <stdio.h>

int main()
{
	// 右寄せ
	printf("###%5d###\n", 123);

	// 左寄せ
	printf("###%-5d###\n", 123);

	// 左寄せの表の作成に便利
	printf("| %-5d |\n", 1);
	printf("| %-5d |\n", 123);
	printf("| %-5d |\n", 12345);
	printf("| %-5d |\n", -123);
	printf("| %-5d |\n", -1);
}
```
