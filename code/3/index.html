    第 3 章 変数とキーボードからの入力¶

    3.2¶

    3.2.1¶

    変数を使わない場合のコード
    #include <stdio.h>

int main()
{
	printf("%.1f\n", ((80 + 100) / 2.0));
}

    3.2.2¶

    変数を使う場合のコード
    #include <stdio.h>

int main()
{
	int mathScore = 80;
	int englishScore = 100;
	double averageScore = (mathScore + englishScore) / 2.0;
	printf("%.1f\n", averageScore);
}

    3.2.4¶

    int 型の変数を定義するコード
    #include <stdio.h>

int main()
{
	// int 型の変数 price を定義する
	int price = 2200;
}
    double 型の変数を定義するコード
    #include <stdio.h>

int main()
{
	// double 型の変数 distance を定義する
	double distance = 42.195;
}

    3.2.5¶

    int 型の変数を使うコード
    #include <stdio.h>

int main()
{
	int price = 2200; // 商品 1 個の価格
	printf("%d 円\n", price); // 1 個買ったときの金額を出力する
	printf("%d 円\n", (price * 10)); // 10 個買ったときの金額を出力する
}
    double 型の変数を使うコード
    #include <stdio.h>

int main()
{
	double distance = 42.195; // マラソン 1 回の距離
	printf("%.3f km\n", distance); // マラソン 1 回の距離を出力する
	printf("%.3f km\n", (distance * 3)); // マラソン 3 回の距離を出力する
}

    3.2.6¶

    複数の変数を定義するコード

    ❌ 誤り: 同じ名前の変数を2 回定義している

int main()

	printf("点数: %d\n", score);
}

    3.5.1¶

    標準入力からの入力を活用するコード
    #include <stdio.h>

int main()
{
	printf("現在の年齢を入力してください > ");
	int age = 0;
	scanf("%d", &age);

	printf("5 年後: %d\n", (age + 5));
	printf("10 年後: %d\n", (age + 10));
}

    3.5.2¶

    標準入力から入力した「分」を「秒」に変換するコード
    #include <stdio.h>

int main()
{
	printf("分を入力してください > ");
	int minutes = 0;
	scanf("%d", &minutes);

	int seconds = minutes * 60;
	printf("%d 分は %d 秒です\n", minutes, seconds);
}

    3.5.3¶

    scanf で値がすぐに代入される変数の初期値を省略するコード
    #include <stdio.h>

int main()
{
	printf("分を入力してください > ");
	int minutes; // 直後に scanf で値を代入するため初期値を省略してよい
	scanf("%d", &minutes);

	int seconds = minutes * 60;
	printf("%d 分は %d 秒です\n", minutes, seconds);
}

int main()

    #include <stdio.h>

int main()
{

    3.8.4¶

    BMI を計算するコード
    #include <stdio.h>

int main()
{
	printf("身長（cm）を入力してください > ");
	double cm;
	scanf("%lf", &cm);

	printf("体重（kg）を入力してください > ");
	double kg;
	scanf("%lf", &kg);

	// 身長をメートルに変換する
	double m = cm / 100.0;

	// BMI を計算して出力する
	double bmi = kg / (m * m);
	printf("BMI: %.2f\n", bmi);
}

    3.8.5¶

    秒を時分秒に変換するコード
    #include <stdio.h>

int main()
{
	printf("秒数を入力してください > ");
	int totalSeconds;
	scanf("%d", &totalSeconds);

	// 時間を計算する
	int hours = totalSeconds / 3600;

	// 分を計算する
	int minutes = (totalSeconds % 3600) / 60;

	// 秒を計算する
	int seconds = totalSeconds % 60;

	// 結果を出力する
	printf("%02d:%02d:%02d\n", hours, minutes, seconds);
}

    3.8.6¶

    税込みの価格を計算するコード
    #include <stdio.h>

int main()
{
	printf("税抜き価格（円）を入力してください > ");
	int price;
	scanf("%d", &price);

	printf("消費税率（%%）を入力してください > ");

	printf("税込み価格: %d 円\n", totalPrice);
}

	{
		int a = 10;
		printf("%d\n", a);
	}

	printf("%d\n", a); // エラー： a のスコープ外
}

    3.10.3¶

    ❌ 誤り: 変数のスコープを広くしたことで起こりうるミスの例
    #include <stdio.h>

int main()
{
	int a, b, c;
	int d, e, f;

	scanf("%d %d %d", &a, &d, &c); // b を使うつもりが d を使ってしまった
	printf("%d\n", (a + d + c)); // b を使うつもりが d を使ってしまった

	scanf("%d %d %d", &b, &e, &f); // d を使うつもりが b を使ってしまった
	printf("%d\n", (a * b * c)); // d, e, f を使うつもりが a, b, c を使ってしまった
}

    シャドーイングの例
    #include <stdio.h>

int main()
{
	int n = 10;
	printf("%d\n", n); // 10

	{
		int n = 20;
		printf("%d\n", n); // 20

		{
			double n = 30;
			printf("%.1f\n", n); // 30.0
		}

		printf("%d\n", n); // 20
	}

	printf("%d\n", n); // 10
}
