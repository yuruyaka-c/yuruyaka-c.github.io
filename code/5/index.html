    第 5 章 計算による変数の値の変更¶

    5.1¶

    5.1.1¶

    複合代入演算子で数を増やしたり減らしたりする
    #include <stdio.h>

int main()
{
	int money = 800;

	money += 200;
	printf("%d\n", money);

	money -= 100;
	printf("%d\n", money);
}
    複合代入演算子を使わない書き方
    #include <stdio.h>

int main()
{
	int money = 800;

	money = (money + 200);
	printf("%d\n", money);

	money = (money - 100);
	printf("%d\n", money);
}

    5.1.2¶

    複合代入演算子で数をかけたり割ったりする
    #include <stdio.h>

int main()
{
	int money = 800;

	money *= 2;
	printf("%d\n", money);

	money /= 10;
	printf("%d\n", money);
}

    5.1.3¶

    剰余代入演算子（%=） の使用例
    #include <stdio.h>

int main()
{
	int money = 1234;
	money %= 1000;
	printf("%d\n", money);
}

    5.1.4¶

    複合代入演算子の右辺に数式を使う
    #include <stdio.h>

int main()
{
	int a = 10, b = 20, c = 30;
	a *= b + c; // 右辺が先に計算される
	printf("%d\n", a);
}
    読みやすい書き方
    #include <stdio.h>

int main()
{
	int a = 10, b = 20, c = 30;
	a *= (b + c);
	printf("%d\n", a);
}

    5.1.5¶

    複合代入演算子の右辺に自身を使う
    #include <stdio.h>

int main()
{
	int score = 20;
	score *= score;
	printf("%d\n", score);

	int money = 1000;
	money += (money + 200);
	printf("%d\n", money);
}

    5.1.6¶

    double 型の変数と複合代入演算子
    #include <stdio.h>

int main()
{
	double distance = 42.195;
	distance += 11.1;
	printf("%f\n", distance);
	distance *= 0.1;
	printf("%f\n", distance);
}

    5.1.7¶

    int と double が混在する複合代入演算（右辺が整数）
    #include <stdio.h>

int main()
{
	double distance = 42.195;
	distance += 50;
	printf("%f\n", distance);
	distance /= 10;
	printf("%f\n", distance);
}

    5.1.8¶

    int と double が混在する複合代入演算（右辺が浮動小数点数）
    #include <stdio.h>

int main()
{
	int score = 10;
	score *= 1.25;
	printf("%d\n", score);
	score -= 0.1;
	printf("%d\n", score);
}

    5.2¶

    5.2.1¶

    複合代入演算の結果を直接使うコード
    #include <stdio.h>

int main()
{
	int money = 1000;
	printf("%d\n", (money += 200));
	printf("%d\n", (money -= 150));
}
    ⭕️ 推奨: 複合代入演算と出力を分けて書くコード
    #include <stdio.h>

int main()
{
	int money = 1000;
	money += 200;
	printf("%d\n", money);
	money -= 150;
	printf("%d\n", money);
}

    5.2.2¶

    代入式の結果を直接使うコード
    #include <stdio.h>

int main()
{
	int money = 1000;
	int result = (money = 2000); // result は 2000 になる

	printf("money: %d\n", money);
	printf("result: %d\n", result);

	printf("%d\n", (money = 3000)); // money は 3000 になり、3000 が出力される
}

    5.3¶

    5.3.1¶

    代入演算子の連続
    #include <stdio.h>

int main()
{
	int a = 10, b = 20, c = 30;

	printf("%d %d %d\n", a, b, c);

	// 代入演算子を連続して使用する
	a = b = c = 40;

	printf("%d %d %d\n", a, b, c);
}

    5.4¶

    5.4.1¶

    前置インクリメント演算子
    #include <stdio.h>

int main()
{
	int age = 20;
	++age;
	printf("%d\n", age);
	++age;
	printf("%d\n", age);
}
    後置インクリメント演算子
    #include <stdio.h>

int main()
{
	int age = 20;
	age++;
	printf("%d\n", age);
	age++;
	printf("%d\n", age);
}

    5.4.2¶

    前置デクリメント演算子
    #include <stdio.h>

int main()
{
	int age = 20;
	--age;
	printf("%d\n", age);
	--age;
	printf("%d\n", age);
}
    後置デクリメント演算子
    #include <stdio.h>

int main()
{
	int age = 20;
	age--;
	printf("%d\n", age);
	age--;
	printf("%d\n", age);
}

    5.4.3¶

    前置と後置の違いを確認するコード
    #include <stdio.h>

int main()
{
	int a = 99;
	printf("%d\n", ++a); // 100（増やした後の値）
	printf("%d\n", a); // 100

	puts("-----");

	int b = 99;
	printf("%d\n", b++); // 99（増やす前の値）
	printf("%d\n", b); // 100
}

    5.4.4¶

    ❌ 誤り: 未定義動作を含むコード
    #include <stdio.h>

int main()
{
	int a = 10;
	a = ++a;
	printf("%d\n", a);

	int b = 10;
	printf("%d %d\n", ++b, ++b);
}
    ❌ 未定義動作を含む実験コード
    #include <stdio.h>

int main()
{
	int b = 10;
	printf("%d %d\n", ++b, ++b);
}
