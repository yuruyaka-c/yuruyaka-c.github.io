# 第 20 章 ファイルの読み書き

## 20.1

### 20.1.1

```c title="ファイルの新規作成"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	// ファイルを開く
	FILE* fp = fopen("test.txt", "w");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	puts("ファイルを開きました。");

	// ファイルを閉じる
	fclose(fp);
	puts("ファイルを閉じました。");
}
```

```c title="ファイルから数値を読み込む"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	FILE* fp = fopen("input.txt", "r");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	puts("ファイルを開きました。");

	int a, b, c;
	fscanf(fp, "%d %d %d", &a, &b, &c);

	printf("%d %d %d\n", (a * 2), (b * 2), (c * 2));

	fclose(fp);
	puts("ファイルを閉じました。");
}
```

### 20.2.2

```c title="fscanf の戻り値を確認しながら数値を読み込む"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	FILE* fp = fopen("input.txt", "r");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	puts("ファイルを開きました。");

	for (;;)
	{
		int n;

		// 整数を 1 個読み込めなかったら終了する
		if (fscanf(fp, "%d", &n) != 1)
		{
			break;
		}

		printf("%d\n", n);
	}

	fclose(fp);
	puts("ファイルを閉じました。");
}
```

```txt title="input.txt の内容"
10 20 30
```

### 20.2.3

```c title="ファイルから 1 文字ずつ読み込む"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	FILE* fp = fopen("input.txt", "r");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	puts("ファイルを開きました。");

	for (;;)
	{
		const int c = fgetc(fp);

		if (c == EOF)
		{
			break;
		}

		printf("%c\n", c);
	}

	fclose(fp);
	puts("ファイルを閉じました。");
}
```

## 20.3

### 20.3.1

```c title="5 個の乱数をファイルに書き出す"
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main()
{
	srand((unsigned int)time(NULL));

	FILE* fp = fopen("numbers.txt", "w");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	puts("ファイルを開きました。");

	for (int i = 0; i < 5; i++)
	{
		const int n = (rand() % 100);
		fprintf(fp, "%d\n", n);
	}

	fclose(fp);
	puts("ファイルを閉じました。");
}
```

### 20.3.2

```c title="ファイルから整数列を読み込んで合計を求める"
#include <stdio.h>
#include <stdlib.h>

int main()
{
	FILE* fp = fopen("numbers.txt", "r");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	puts("ファイルを開きました。");
	long long sum = 0;

	for (;;)
	{
		int n;
		if (fscanf(fp, "%d", &n) != 1)
		{
			break;
		}

		sum += n;
	}

	fclose(fp);
	puts("ファイルを閉じました。");
	printf("合計: %lld\n", sum);
}
```

```txt title="numbers.txt の内容例"
86
45
92
6
17
```

## 20.4

### 20.4.2

```c title="バイナリデータを書き込む"
#include <stdio.h>
#include <stdlib.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

int main()
{
	const int a[] = { 10, 20, 30, 40, 50 };

	FILE* fp = fopen("data.bin", "wb");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	fwrite(a, sizeof(int), COUNTOF(a), fp);
	puts("バイナリデータを書き込みました。");

	fclose(fp);
}
```

```txt title="data.bin の内容例（十六進表記）"
0A 00 00 00 14 00 00 00 1E 00 00 00 28 00 00 00 32 00 00 00
```

### 20.4.3

```c title="バイナリデータを読み込む"
#include <stdio.h>
#include <stdlib.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

int main()
{
	int a[5] = {};

	FILE* fp = fopen("data.bin", "rb");
	if (fp == NULL)
	{
		puts("ファイルを開けませんでした。");
		return EXIT_FAILURE;
	}

	fread(a, sizeof(int), COUNTOF(a), fp);
	puts("バイナリデータを読み込みました。");

	fclose(fp);

	for (int i = 0; i < COUNTOF(a); i++)
	{
		printf("%d ", a[i]);
	}

	putchar('\n');
}
```
