    第 2 章 基本的な計算と結果の出力¶

    2.1¶

    計算式を文字列の中に書いているコード
    #include <stdio.h>

int main()
{
	puts("1 + 2");
}

    2.1.1¶

    整数と整数の式がいくつか書かれているコード
    #include <stdio.h>

int main()
{
	3;
	100 + 50;
	-50 - 20;
}

    2.1.2¶

    ❌ 誤り: 整数や整数の式を puts 関数に渡すコード
    #include <stdio.h>

int main()
{
	puts(3);
	puts(100 + 50);
	puts(-50 - 20);
}

    2.1.3¶

    整数を printf 関数で出力するコード①
    #include <stdio.h>

int main()
{
	printf("%d 円\n", 100 + 50);
}

    2.1.4¶

    整数を printf 関数で出力するコード②
    #include <stdio.h>

int main()
{
	printf("%d 回目のテストを受けた\n", 3);
	printf("結果は %d 点だった\n", 40 + 60);
}

    2.1.5¶

    printf が文字列の末尾を自動で改行しないことを確認するコード
    #include <stdio.h>

int main()
{
	printf("%d 回目のテストを受けた", 3);
	printf("結果は %d 点だった", 40 + 60);
}

    2.1.6¶

    文字列の中で複数の書式指定子を使うコード
    #include <stdio.h>

int main()
{
	printf("数学は %d 点、英語は %d 点\n", 30, 50 + 50);
}

    2.1.7¶

    ❌ 誤り: 書式指定子の個数と渡す整数の個数が一致しない
    #include <stdio.h>

int main()
{
	printf("数学は %d 点、英語は %d 点\n", 10 + 20);
}

    2.1.8¶

    ❌ 誤り: int 型の範囲外の整数を %d で出力しようとする
    #include <stdio.h>

int main()
{
	printf("%d\n", 111222333444555);
}

    2.2¶

    2.2.1¶

    整数どうしの足し算と引き算の結果を出力するコード
    #include <stdio.h>

int main()
{
	printf("%d\n", 1 + 2);
	printf("%d\n", 10 - 3);
	printf("%d\n", -50 + 70 - 10);
}

    2.2.2¶

    整数どうしのかけ算と割り算の結果を出力するコード
    #include <stdio.h>

int main()
{
	printf("%d\n", 5 * 10);
	printf("%d\n", 60 / 2);
	printf("%d\n", 10 * 4 / 2);
}

    2.2.3¶

    整数どうしの割り切れない割り算の結果を出力するコード
    #include <stdio.h>

int main()
{
	printf("%d\n", 10 / 3);
	printf("%d\n", 22 / 5);
}

    2.2.4¶

    整数どうしの剰余算の結果を出力するコード
    #include <stdio.h>

int main()
{
	printf("%d\n", 10 % 3);
	printf("%d\n", 22 % 5);
}

    2.2.5¶

    剰余算の結果を確認するコード
    #include <stdio.h>

int main()
{
	printf("%d\n", 10 % 3);
	printf("%d\n", 10 % -3);
	printf("%d\n", -10 % 3);
	printf("%d\n", -10 % -3);
}

    2.2.6¶

    190 分を「時間」と「分」に分けるコード
    #include <stdio.h>

int main()
{
	printf("%d 時間 %d 分\n", 190 / 60, 190 % 60);
}
    各位の数を取り出すコード
    #include <stdio.h>

int main()
{
	printf("千の位: %d\n", 3456 / 1000 % 10);
	printf("百の位: %d\n", 3456 / 100 % 10);
	printf("十の位: %d\n", 3456 / 10 % 10);
	printf("一の位: %d\n", 3456 / 1 % 10);
}

    2.2.7¶

    丸かっこで演算子の優先順位を変えて計算するコード
    #include <stdio.h>

int main()
{
	printf("%d\n", 10 + 2 * 2);
	printf("%d\n", (10 + 2) * 2);
	printf("%d\n", (2 * (1 + 1) + 2) * 2);
}

    2.2.8¶

    読みやすさのために丸かっこを使うコード
    #include <stdio.h>

int main()
{
	printf("%d %d\n", 10 + 2 * 2, 5 + 8 + 7);
	printf("%d %d\n", (10 + 2 * 2), (5 + 8 + 7));
}

    2.3¶

    2.3.1¶

    単一行コメントの例
    #include <stdio.h> // printf を使うために必要

// main 関数
int main()
{
	// かけ算が先に計算される
	printf("%d\n", (10 + 2 * 2)); // 14
}
    コメントアウトの例
    #include <stdio.h>

int main()
{
	//puts("Apple");
	puts("Banana");
}

    2.3.2¶

    範囲コメントの例
    #include <stdio.h>

int main()
{
	/* この範囲はコメントになる */
	puts("Apple");
	/*
	この範囲もコメントになる
	puts("Banana");
	*/
	printf("%d\n", /*123*/ 456);
}

    2.3.3¶

    ❌ 誤り: 範囲コメントをネストしている
    #include <stdio.h>

int main()
{
	/*
	/* この範囲はコメントになる */
	puts("Apple");
	*/
}
    範囲コメントの中に単一行コメントがあるコード
    #include <stdio.h>

int main()
{
	/*
	// この範囲はコメントになる
	puts("Apple");
	*/
}

    2.3.4¶

    文字列の中に「//」や「/*」があるコード
    #include <stdio.h>

int main()
{
	puts("Apple // Banana");
	puts("/* Apple */");
}

    2.3.5¶

    コードに説明を書く例
    #include <stdio.h>

int main()
{
	// 3 日間が何秒であるかを計算する（1 日は 86400 秒）
	printf("%d\n", (3 * 86400));
}

    2.3.6¶

    コメントアウト機能を試すコード
    #include <stdio.h>

int main()
{
	puts("Apple");
	puts("Banana");
	puts("Cat");
	puts("Dog");

    2.4.1¶

    浮動小数点数と浮動小数点数の式が書かれているコード
    #include <stdio.h>

int main()
{
	1.5;
	3.14 + 1.0;
}

    2.4.2¶

    ❌ 誤り: 浮動小数点数や浮動小数点数の式を puts 関数に渡している
    #include <stdio.h>

int main()
{
	puts(1.5);
	puts(3.5 + 0.5);
}
    ❌ 誤り: 浮動小数点数を整数用の書式指定子 %d で出力している
    #include <stdio.h>

int main()
{
	printf("%d\n", 1.5);
	printf("%d\n", (3.5 + 0.5));
}

    2.4.3¶

    浮動小数点数を printf で出力するコード
    #include <stdio.h>

int main()
{
	printf("%f\n", 1.5);
	printf("%f\n", (3.5 + 0.5));
	printf("%f\n", -0.0012);
}

    2.4.4¶

    小数点以下の桁数を指定して出力するコード
    #include <stdio.h>

int main()
{
	printf("円周率は %f\n", 3.141592653589793);
	printf("円周率は %.10f\n", 3.141592653589793);
	printf("円周率は %.4f\n", 3.141592653589793);
	printf("円周率は %.2f\n", 3.141592653589793);
	printf("円周率は %.0f\n", 3.141592653589793);
}
    最近接偶数への丸めを確認するコード
    #include <stdio.h>

int main()
{
	printf("%.0f\n", 127.5);
	printf("%.0f\n", 128.5);
	printf("%.0f\n", 129.5);
	printf("%.0f\n", 130.5);
	puts("----");
	printf("%.1f\n", 1.25);
	printf("%.1f\n", 2.25);
	printf("%.1f\n", 1.35);
	printf("%.1f\n", 2.35);
}

    2.4.5¶

    浮動小数点数の精度の限界を確認するコード
    #include <stdio.h>

int main()
{
	printf("%.25f\n", 11111.1111111111111111111111111111);
	printf("%.25f\n", 0.1);
	printf("%.25f\n", 1.5);
}

    2.5¶

    2.5.1¶

    浮動小数点数の計算を行うコード
    #include <stdio.h>

int main()
{
	printf("%f\n", (1.5 + 0.25));
	printf("%f\n", (-3.14 - 1.0));
	printf("%f\n", (0.1 * 0.1));
	printf("%f\n", (10.0 / 3.0));
}
    ❌ 誤り: 浮動小数点数に「%」演算子を使っている
    #include <stdio.h>

int main()
{
	printf("%f\n", (10.0 % 3.0));
}

    2.5.2¶

    整数と浮動小数点数が交ざる計算を行うコード
    #include <stdio.h>

int main()
{
	printf("%f\n", (1.5 + 2));
	printf("%f\n", (1 + 0.5));
}

    2.5.3¶

    整数どうしの計算になる部分に注意が必要なコード
    #include <stdio.h>

int main()
{
	printf("%f\n", (1 / 2 + 0.1)); // 整数どうしの 1 / 2 は 0 になる
	printf("%f\n", (0.1 + 5 / 2)); // 整数どうしの 5 / 2 は 2 になる
	printf("%f\n", (0.1 * 2 / 10)); // 0.2 / 10 なので 0.02 になる
}
    浮動小数点数として計算するコード
    #include <stdio.h>

int main()
{
	printf("%f\n", (1.0 / 2 + 0.1));
	printf("%f\n", (0.1 + 5 / 2.0));
}

    2.6¶

    2.6.2¶

    書式指定子を含まないフォーマット文字列の例
    #include <stdio.h>

int main()
{
	printf("App");
	printf("le\n");
}

    2.6.3¶

    フォーマット文字列のエスケープシーケンス %% を使うコード
    #include <stdio.h>

int main()
{
	printf("80%%\n");
	printf("%d%%\n", 90);

	// puts に渡す文字列はフォーマット文字列ではないため、
	// % の出力にエスケープシーケンスは不要
	puts("100%");
}

    2.6.4¶

    ❌ 誤り: %% を使わなかった場合の事故の例
    #include <stdio.h>

int main()
{
	printf("50%dog! 100%frog!\n"); // %d や %f が書式指定子として解釈される
}
    修正したコード
    #include <stdio.h>

int main()
{
	printf("50%%dog! 100%%frog!\n"); // %% を使って % を出力する
	puts("50%dog! 100%frog!"); // または puts を使う
}

    2.7¶

    2.7.1¶

    フィールド幅を指定するコード
    #include <stdio.h>

int main()
{
	printf("%5d\n", 1);
	printf("%5d\n", 123);
	printf("%5d\n", 12345);
	printf("%5d\n", -123);
	printf("%5d\n", -1);
}

    2.7.2¶

    フィールド幅を超える出力をするコード
    #include <stdio.h>

int main()
{
	printf("%5d\n", 123456);
	printf("%5d\n", -123456);
	printf("%5d\n", 1234567890);
	printf("%5d\n", -1234567890);
}

    2.7.3¶

    ゼロ埋めを指定するコード
    #include <stdio.h>

int main()
{
	printf("%05d\n", 1);
	printf("%05d\n", 123);
	printf("%05d\n", 12345);

	// 時刻の出力に便利
	printf("%02d:%02d:%02d\n", 1, 2, 0);
	printf("%02d:%02d:%02d\n", 12, 23, 34);
}

    2.7.4¶

    左寄せをするコード
    #include <stdio.h>

int main()
{
	// 右寄せ
	printf("###%5d###\n", 123);

	// 左寄せ
	printf("###%-5d###\n", 123);

	// 左寄せの表の作成に便利
	printf("| %-5d |\n", 1);
	printf("| %-5d |\n", 123);
	printf("| %-5d |\n", 12345);
	printf("| %-5d |\n", -123);
	printf("| %-5d |\n", -1);
}
