    18.1¶

    18.1.1¶

    void* にさまざまな型のアドレスを代入する例
    #include <stdio.h>

int main()
{
	int a = 10;
	double b = 3.14;

	void* p;

	p = &a; // int* → void*
	printf("%p\n", p);

	p = &b; // double* → void*
	printf("%p\n", p);
}

    18.1.2¶

    ❌ 誤り: void* を直接使おうとしている
    #include <stdio.h>

int main()
{
	int a = 10;
	void* p = &a;
	printf("%d\n", *p); // ❌️コンパイルエラー
}

    18.1.3¶

    void* をキャストして使う例
    #include <stdio.h>

int main()
{
	int a = 10;
	void* p = &a;
	printf("%d\n", *(int*)p); // void* を int* にキャストして間接参照
}

    18.2.2¶

    int 型の比較関数の例
    #include <stdio.h>

int CompareInt(const void* p1, const void* p2)
{
	const int a = *(const int*)p1;
	const int b = *(const int*)p2;

	if (a < b)
	{
		return -1;
	}
	else if (a > b)
	{
		return 1;
	}
	else
	{
		return 0;
	}
}

int main()
{
	int x = 10;
	int y = 20;
	printf("比較結果: %d\n", CompareInt(&x, &y));
}

    18.2.4¶

    ⭕️ 正しい実装と避けるべき実装の比較
    #include <stdio.h>

// ⭕️正しい比較関数
int CompareInt(const void* p1, const void* p2)
{
	const int a = *(const int*)p1;
	const int b = *(const int*)p2;
	return (a > b) - (a < b);
}

// ⚠️避けるべき比較関数
int CompareInt2(const void* p1, const void* p2)
{
	const int a = *(const int*)p1;
	const int b = *(const int*)p2;
	return a - b; // ⚠️整数オーバーフローの可能性あり
}

int main()
{
	int x = 2000000000;
	int y = -2000000000;
	printf("CompareInt の結果: %d\n", CompareInt(&x, &y));
	printf("CompareInt2 の結果: %d\n", CompareInt2(&x, &y));
}

    18.2.5¶

    qsort を使った int 型配列のソート例（昇順）
    #include <stdio.h>
#include <stdlib.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

void PrintIntArray(const int arr[], int count)
{
	putchar('[');

	for (int i = 0; i < count; ++i)
	{
		printf(((i == 0) ? "%d" : ", %d"), arr[i]);
	}

	puts("]");
}

int CompareInt(const void* p1, const void* p2)
{
	const int a = *(const int*)p1;
	const int b = *(const int*)p2;
	return (a > b) - (a < b);
}

int main()
{
	int values[] = { 30, 8, 24, 21, 55, 79, 92, 6 };
	qsort(values, COUNTOF(values), sizeof(int), CompareInt);
	PrintIntArray(values, COUNTOF(values));
}

    18.2.8¶

    qsort を使った構造体配列のソート例
    #include <stdio.h>
#include <stdlib.h>
#include <math.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

typedef struct
{
	double x;
	double y;
} Vec2;

// Vec2 の原点からの距離を比較する関数
int CompareVec2(const void* p1, const void* p2)
{
	const Vec2* v1 = (const Vec2*)p1;
	const Vec2* v2 = (const Vec2*)p2;
	const double len1 = hypot(v1->x, v1->y);
	const double len2 = hypot(v2->x, v2->y);
	return (len1 > len2) - (len1 < len2);
}

int main()
{
	Vec2 vectors[] =
	{
		{ 3.0, 4.0 },
		{ 1.0, 1.0 },
		{ -1.0, 5.0 },
		{ 2.0, 2.0 }
	};

	qsort(vectors, COUNTOF(vectors), sizeof(Vec2), CompareVec2);

	// ソート結果の出力
	for (int i = 0; i < COUNTOF(vectors); ++i)
	{
		printf("(%.2f, %.2f)\n", vectors[i].x, vectors[i].y);
	}
}

    18.3.1¶

    スキージャンプの飛型点を計算するプログラム
    #include <stdio.h>
#include <stdlib.h>

int CompareDouble(const void* p1, const void* p2)
{
	const double a = *(const double*)p1;
	const double b = *(const double*)p2;
	return (a > b) - (a < b);
}

int main()
{
	printf("5 人の審判の点数を入力してください > ");
	double scores[5];

	for (int i = 0; i < 5; ++i)
	{
		scanf("%lf", &scores[i]);
	}

	// 点数を昇順にソート
	qsort(scores, 5, sizeof(double), CompareDouble);

	double total = 0.0;

	// 最低点 scores[0] と 最高点 scores[4] を除いた 3 つを合計
	for (int i = 1; i < (5 - 1); ++i)
	{
		total += scores[i];
	}

	printf("飛型点: %.1f\n", total);
}

    18.3.2¶

    成績順に学生をソートするプログラム
    #include <stdio.h>
#include <stdlib.h>
#include <string.h>
#define COUNTOF(arr) ((int)(sizeof(arr)/sizeof((arr)[0])))

typedef struct
{
	char name[21];
	int score;
} Student;

int CompareStudentByScoreDesc(const void* p1, const void* p2)
{
	const Student* s1 = (const Student*)p1;
	const Student* s2 = (const Student*)p2;

	if (s1->score != s2->score)
	{
		return (s1->score < s2->score) - (s1->score > s2->score);
	}

	return strcmp(s1->name, s2->name); // 同点の場合は名前の辞書順
}

int main()
{
	Student students[] =
	{
		{ "Alice", 85 },
		{ "Bob", 92 },
		{ "Carol", 78 },
		{ "Dave", 92 },
		{ "Eve", 88 }
	};

	qsort(students, COUNTOF(students), sizeof(Student), CompareStudentByScoreDesc);

	printf("順位   名前    点数\n");
	printf("-----------------\n");

	for (int i = 0; i < COUNTOF(students); ++i)
	{
		printf("%2d 位 %-6s %3d\n", (i + 1), students[i].name, students[i].score);
	}
}
