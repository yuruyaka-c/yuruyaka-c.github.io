    整数の絶対値を求める例
    #include <stdio.h>
#include <stdlib.h>

int main()
{
	int a;
	scanf("%d", &a);
	printf("%d\n", abs(a));

	scanf("%lld", &b);
	printf("%lld\n", llabs(b));
}

    戻り値を演算や条件式で使う例
    #include <stdio.h>
#include <stdlib.h>

int main()
{
	int x;
	scanf("%d", &x);

	if (10 < abs(x))
	{

int main()

    10.3¶

    10.3.2¶

    関数定義の例
    #include <stdio.h>

int Square(int n)
{
	return n * n;
}

int main()
{
	printf("%d\n", Square(5));
}
    関数宣言を使った関数定義の例
    #include <stdio.h>

int Square(int n); // 関数宣言

int main()
{
	printf("%d\n", Square(5));
}

int Square(int n) // 関数定義
{
	return n * n;
}
    誤り: 関数宣言や定義の前で関数を呼んでいる
    #include <stdio.h>

int main()
{
	// 誤り：関数宣言より前で呼び出している
	printf("%d\n", Square(5));
}

int Square(int n);

int Square(int n)
{
	return n * n;
}

    10.3.4¶

    仮引数と実引数の例
    #include <stdio.h>

int Square(int n); // n は仮引数

int main()
{
	int result = Square(5); // 5 は実引数
	printf("%d\n", result); // "%d\n" と result は printf の実引数
}

int Square(int n) // n は仮引数
{
	return n * n;
}

    10.3.5¶

    return 文の挙動を確認するコード
    #include <stdio.h>

int Square(int n)
{
	puts("A");
	return n * n;
	puts("B"); // ここは実行されない
}

int main()
{
	int result = Square(5);
	printf("%d\n", result);
}
    ❌ 誤り: 戻り値の型に合わない値をreturn している
    #include <stdio.h>

int Half(int n) // 戻り値の型は int
{
	return n / 2.0; // 計算結果は double 型だが、int に変換されて返される
}

int main()
{
	printf("%d\n", Half(5));
}
    ❌ 誤り: return 文が実行されない場合がある
    #include <stdio.h>

int Max(int a, int b)
{
	if (a > b)
	{
		return a;
	}

	if (a < b)
	{
		return b;
	}

	// ❌️ a == b の場合に return 文が実行されない
}

int main()
{
	printf("%d\n", Max(3, 3));
}
    ⭕️ 必ず return 文が実行される
    #include <stdio.h>

int Max(int a, int b)
{
	if (a > b)
	{
		return a;
	}
	else
	{
		return b;
	}
}

int main()
{
	printf("%d\n", Max(3, 3));
}

    10.3.7¶

    戻り値のない関数での return 文の例
    #include <stdio.h>

void DrawLine(int length)
{
	if (length <= 0) // 長さが 0 以下の場合は何もせず終了
	{
		return;
	}

	if (100 < length) // 長さが大きすぎる場合は何もせず終了
	{
		return;
	}

	for (int i = 0; i < length; ++i)
	{
		putchar('-');
	}

	putchar('\n');
}

int main()
{
	DrawLine(0); // 何も出力されない
	DrawLine(200); // 何も出力されない
	DrawLine(10);
}

    10.3.8¶

    関数本体の変数のスコープ
    #include <stdio.h>

int Square(int n) // 仮引数 n のスコープはこの関数本体内
{
	int result = n * n; // result のスコープはこの関数本体内
	return result;
}

int main()
{
	int n = 2; // Square 関数の仮引数 n とは無関係
	printf("%d\n", Square(n + 3));

	// ❌️Square 関数内の result は main 関数内では使えない
	// printf("%d\n", result);
}
    関数内の変数の挙動を確認するコード
    #include <stdio.h>

int Counter()
{
	int count = 0; // count は毎回新しく作られる
	++count;
	return count;
}

int main()
{
	printf("%d\n", Counter());
	printf("%d\n", Counter());
	printf("%d\n", Counter());
}

    10.4.3¶

    指定した幅・高さ・文字で長方形を描く関数
    #include <stdio.h>

void DrawRectangle(int width, int height, char c)
{
	for (int y = 0; y < height; ++y)
	{
		for (int x = 0; x < width; ++x)
		{
			putchar(c);
		}

		putchar('\n');
	}
}

int main()
{
	DrawRectangle(4, 3, '#');
	DrawRectangle(6, 6, '*');
}

    10.4.4¶

    2 つの整数のうち大きい方を返す関数
    #include <stdio.h>

int Max(int a, int b)
{
	if (a > b)
	{
		return a;
	}
	else
	{
		return b;
	}

int main()

    条件演算子を使った Max 関数の例
    #include <stdio.h>

int Max(int a, int b)
{
	return (a > b) ? a : b;
}

int main()

    10.4.5¶

    3 つの整数のうち最大値を返す関数
    #include <stdio.h>

int Max3(int a, int b, int c)
{
	int max = a;

	if (max < b)
	{
		max = b;
	}

	if (max < c)
	{
		max = c;
	}

	return max;
}

int main()
{
	int x, y, z;
	scanf("%d %d %d", &x, &y, &z);
	printf("%d\n", Max3(x, y, z));
}
    Max 関数を使った Max3 関数の例
    #include <stdio.h>

int Max(int a, int b)
{
	return (a > b) ? a : b;
}

int Max3(int a, int b, int c)
{
	return Max(Max(a, b), c);
}

int main()
{
	int x, y, z;
	scanf("%d %d %d", &x, &y, &z);
	printf("%d\n", Max3(x, y, z));
}

    2 点間の距離を求める関数

#include <math.h>

double Distance(double x1, double y1, double x2, double y2)

int main()

}

    10.4.7¶

    素数判定を行う関数
    #include <stdio.h>

bool IsPrime(int n)
{
	if (n <= 1)
	{
		return false;
	}

	for (int i = 2; i <= (n / 2); ++i)
	{
		if ((n % i) == 0) // 割り切れた場合
		{
			return false;
		}
	}

	return true;
}

int main()
{
	int x;
	scanf("%d", &x);

	if (IsPrime(x))
	{
		puts("素数");
	}
	else
	{
		puts("素数ではない");
	}
}
